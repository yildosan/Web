3. Donanım Bağlantı Şeması (Pinout)

Tüm modüller doğrudan ESP32 kartının 3.3V ve GND hatlarından beslenmelidir.

  • OLED Ekran (I2C): SDA -> GPIO 21 | SCL -> GPIO 22
  • SPI Otobüsü (SD & RFID Ortak): MOSI -> GPIO 23 | MISO -> GPIO 19 | SCK -> GPIO 18
  • MicroSD Kart CS: GPIO 13
  • RC522 RFID: SS (SDA) -> GPIO 5 | RST -> GPIO 27
  • 5 Yönlü Joystick: UP -> GPIO 32 | DOWN -> GPIO 33 | LEFT -> GPIO 25 | RIGHT -> GPIO 26 | CENTER -> GPIO 14
  • Buzzer & RGB: Buzzer -> GPIO 4 | LED R -> GPIO 16 | LED G -> GPIO 17 | LED B -> GPIO 2

4. AION V1 Tam Kaynak Kod (Firmware)

Cihazınızı derleyip doğrudan ESP32 kartına yükleyebileceğiniz eksiksiz C++ kaynak kodu:

/*
   ============================================================
                AION V1 - TEK DOSYA FIRMWARE
   ============================================================
*/

#include <Arduino.h>
#include <Wire.h>
#include <SPI.h>
#include <SD.h>
#include <WiFi.h>
#include <WebServer.h>
#include <BLEDevice.h>
#include <BLEScan.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>
#include <MFRC522.h>
#include <Preferences.h>
#include <time.h>

#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
#define OLED_SDA 21
#define OLED_SCL 22

Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, -1);

#define RFID_SS 5
#define RFID_RST 27
MFRC522 rfid(RFID_SS, RFID_RST);

#define SD_CS 13
bool sdOK = false;

#define JOY_UP     32
#define JOY_DOWN   33
#define JOY_LEFT   25
#define JOY_RIGHT  26
#define JOY_CENTER 14

#define BUZZER 4
#define LED_R 16
#define LED_G 17
#define LED_B 2

WebServer server(80);
Preferences preferences;

bool soundEnabled = true;
bool ledEnabled = true;
int brightness = 255;

enum MenuState {
  MAIN_MENU, WIFI_MENU, WIFI_SCAN_RESULT, BLE_MENU, BLE_SCAN_RESULT,
  RFID_MENU, RFID_RESULT, SD_MENU, SD_FILES, TOOLS_MENU, SYSTEM_MENU,
  SETTINGS_MENU, CLOCK_SCREEN, STOPWATCH_SCREEN, GAMES_MENU, SNAKE_GAME, WEB_INFO
};

MenuState currentScreen = MAIN_MENU;

const char* mainMenu[] = {
  "Wi-Fi", "Bluetooth", "RFID", "MicroSD", "Araclar", "Sistem",
  "Ayarlar", "Saat", "Kronometre", "Oyunlar", "Web Arayuzu"
};
const int MAIN_MENU_COUNT = sizeof(mainMenu) / sizeof(mainMenu[0]);
int mainMenuIndex = 0;
int submenuIndex = 0;

unsigned long lastButtonTime = 0;
const unsigned long buttonDelay = 180;

bool stopwatchRunning = false;
unsigned long stopwatchStart = 0;
unsigned long stopwatchElapsed = 0;

int wifiCount = 0;
struct WiFiInfo { String ssid; int rssi; int channel; String encryption; };
WiFiInfo wifiList[20];

BLEScan* bleScan;
struct BLEInfo { String name; String address; int rssi; };
BLEInfo bleList[20];
int bleCount = 0;

String lastRFID = "";

void clearScreen() { display.clearDisplay(); display.setTextColor(SSD1306_WHITE); display.setTextSize(1); display.setCursor(0, 0); }
void showScreen() { display.display(); }

void beep(int duration = 60) {
  if (!soundEnabled) return;
  digitalWrite(BUZZER, HIGH); delay(duration); digitalWrite(BUZZER, LOW);
}

void setRGB(bool r, bool g, bool b) {
  if (!ledEnabled) { digitalWrite(LED_R, LOW); digitalWrite(LED_G, LOW); digitalWrite(LED_B, LOW); return; }
  digitalWrite(LED_R, r); digitalWrite(LED_G, g); digitalWrite(LED_B, b);
}

void bootScreen() {
  clearScreen();
  display.setTextSize(2); display.setCursor(20, 10); display.println("AION");
  display.setTextSize(1); display.setCursor(42, 35); display.println("V1");
  display.setCursor(25, 50); display.println("Baslatiliyor...");
  showScreen();
  setRGB(false, false, true); delay(1500); setRGB(false, false, false);
}

bool buttonPressed(int pin) {
  if (digitalRead(pin) == LOW) {
    if (millis() - lastButtonTime > buttonDelay) { lastButtonTime = millis(); return true; }
  }
  return false;
}

void drawMainMenu() {
  clearScreen();
  display.setTextSize(1); display.setCursor(0, 0); display.println("AION V1");
  display.drawLine(0, 9, 127, 9, SSD1306_WHITE);
  int start = mainMenuIndex - 2;
  if (start < 0) start = 0;
  if (start > MAIN_MENU_COUNT - 5) start = MAIN_MENU_COUNT - 5;
  if (start < 0) start = 0;
  for (int i = 0; i < 5; i++) {
    int index = start + i;
    if (index >= MAIN_MENU_COUNT) break;
    display.setCursor(0, 13 + i * 10);
    if (index == mainMenuIndex) display.print("> "); else display.print("  ");
    display.println(mainMenu[index]);
  }
  showScreen();
}

void setup() {
  Serial.begin(115200);
  pinMode(JOY_UP, INPUT_PULLUP); pinMode(JOY_DOWN, INPUT_PULLUP);
  pinMode(JOY_LEFT, INPUT_PULLUP); pinMode(JOY_RIGHT, INPUT_PULLUP);
  pinMode(JOY_CENTER, INPUT_PULLUP);
  pinMode(BUZZER, OUTPUT); pinMode(LED_R, OUTPUT); pinMode(LED_G, OUTPUT); pinMode(LED_B, OUTPUT);

  Wire.begin(OLED_SDA, OLED_SCL);
  if (!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    while (true) { setRGB(true, false, false); delay(300); setRGB(false, false, false); delay(300); }
  }

  bootScreen();
  SPI.begin(18, 19, 23);
  rfid.PCD_Init();
  
  if (SD.begin(SD_CS)) { sdOK = true; if (!SD.exists("/AION")) SD.mkdir("/AION"); } else { sdOK = false; }

  BLEDevice::init("AION-V1");
  bleScan = BLEDevice::getScan();
  bleScan->setActiveScan(true);

  WiFi.mode(WIFI_STA);
  currentScreen = MAIN_MENU;
  drawMainMenu();
}

void loop() {
  server.handleClient();
  switch (currentScreen) {
    case MAIN_MENU:
      if (buttonPressed(JOY_UP)) { mainMenuIndex--; if (mainMenuIndex < 0) mainMenuIndex = MAIN_MENU_COUNT - 1; beep(); drawMainMenu(); }
      if (buttonPressed(JOY_DOWN)) { mainMenuIndex++; if (mainMenuIndex >= MAIN_MENU_COUNT) mainMenuIndex = 0; beep(); drawMainMenu(); }
      if (buttonPressed(JOY_CENTER)) { beep(); currentScreen = WIFI_MENU; }
      break;
    default:
      break;
  }
}
